# Файл: queries/recommend/01_exercises_for_patient.rq
# id: exercises_for_patient
# title: Упражнения для пациента
# button: Подобрать упражнения
# description: Упражнения, показанные при диагнозе пациента или работающие на его цель реабилитации. Исключаются противопоказанные при диагнозе, слишком сложные (сложность > подготовка + 1; при ожирении или возрасте 60+ — не выше 2) и с приостановленным испытанием. Рейтинг = 0,6 × эффективность + 8 × балл безопасности (0–100).
# param: пациент | individual:Пациент | reh:Пациент_Новиков
# card: title=упражнение; score=рейтинг; fields=почему,сложность,эффективность,безопасность,инвентарь,статусИспытания
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?у ?упражнение ?рейтинг ?эффективность ?безопасность ?сложность ?инвентарь ?статусИспытания
       (GROUP_CONCAT(DISTINCT ?причина; SEPARATOR="; ") AS ?почему)
WHERE {
  VALUES ?пациент { reh:Пациент_Новиков }   # PARAM пациент
  ?пациент reh:имеетДанные/reh:содержитДиагноз ?диагноз ;
           reh:уровеньПодготовки ?подготовка ;
           reh:возраст ?возраст ;
           reh:весКг ?вес ;
           reh:ростСм ?рост .
  {   # путь 1: показание упражнения подходит к диагнозу
    ?показание reh:показаноПри ?диагноз ;
               reh:описание ?причина .
    ?у reh:имеетПоказание ?показание .
  } UNION {   # путь 2: упражнение работает на цель, которую требует диагноз
    ?диагноз reh:требуетЦели ?цель .
    ?цель reh:название ?причина .
    ?у reh:влияетНа ?цель .
  }
  ?у reh:название ?упражнение ;
     reh:уровеньСложности ?сложность ;
     reh:основаноНаИспытании ?испытание .
  ?испытание reh:показываетЭффективность/reh:процентЭффективности ?эффективность ;
             reh:определяетУровеньБезопасности/reh:баллБезопасности ?безопасность ;
             reh:имеетСтатус/reh:название ?статусИспытания .
  OPTIONAL { ?у reh:требуетИнвентарь/reh:название ?инвентарь }

  FILTER NOT EXISTS { ?у reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз }
  BIND (?вес / ((?рост / 100.0) * (?рост / 100.0)) AS ?имт)
  BIND (IF(?имт >= 30 || ?возраст >= 60, 2, ?подготовка + 1) AS ?максСложность)
  FILTER (?сложность <= ?максСложность)
  FILTER (?статусИспытания != "Приостановлено")
  BIND (0.6 * ?эффективность + 8 * ?безопасность AS ?рейтинг)
}
GROUP BY ?у ?упражнение ?рейтинг ?эффективность ?безопасность ?сложность ?инвентарь ?статусИспытания
ORDER BY DESC(?рейтинг) ?сложность

# Файл: queries/recommend/02_exercises_for_diagnosis.rq
# id: exercises_for_diagnosis
# title: Упражнения по диагнозу
# button: Что делать при диагнозе
# description: Без привязки к пациенту: какие упражнения рекомендованы при выбранном диагнозе (по показаниям и по цели реабилитации), без противопоказанных и без приостановленных испытаний. Сортировка по рейтингу.
# param: диагноз | individual:Диагноз | reh:Диагноз_Плоскостопие
# card: title=упражнение; score=рейтинг; fields=почему,сложность,эффективность,безопасность,инвентарь
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?у ?упражнение ?рейтинг ?эффективность ?безопасность ?сложность ?инвентарь
       (GROUP_CONCAT(DISTINCT ?причина; SEPARATOR="; ") AS ?почему)
WHERE {
  VALUES ?диагноз { reh:Диагноз_Плоскостопие }   # PARAM диагноз
  {
    ?показание reh:показаноПри ?диагноз ;
               reh:описание ?причина .
    ?у reh:имеетПоказание ?показание .
  } UNION {
    ?диагноз reh:требуетЦели ?цель .
    ?цель reh:название ?причина .
    ?у reh:влияетНа ?цель .
  }
  ?у reh:название ?упражнение ;
     reh:уровеньСложности ?сложность ;
     reh:основаноНаИспытании ?испытание .
  ?испытание reh:показываетЭффективность/reh:процентЭффективности ?эффективность ;
             reh:определяетУровеньБезопасности/reh:баллБезопасности ?безопасность ;
             reh:имеетСтатус/reh:название ?статус .
  OPTIONAL { ?у reh:требуетИнвентарь/reh:название ?инвентарь }
  FILTER NOT EXISTS { ?у reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз }
  FILTER (?статус != "Приостановлено")
  BIND (0.6 * ?эффективность + 8 * ?безопасность AS ?рейтинг)
}
GROUP BY ?у ?упражнение ?рейтинг ?эффективность ?безопасность ?сложность ?инвентарь
ORDER BY DESC(?рейтинг) ?сложность

# Файл: queries/recommend/03_program_for_patient.rq
# id: program_for_patient
# title: Готовая программа для пациента
# button: Порекомендовать программу
# description: Ищет уже существующие программы других пациентов как шаблон. Совпадение 3 — тот же диагноз, 2 — та же цель, 1 — цель того же вида. Для каждой программы считается, сколько её упражнений не подходят этому пациенту (противопоказаны или слишком сложны). Лучшие — с высоким совпадением и без неподходящих упражнений.
# param: пациент | individual:Пациент | reh:Пациент_Новиков
# card: title=программа; score=совпадение; warn=неПодходит; fields=пациентПрограммы,специалист,длительностьНедель,всегоУпражнений,средняяЭффективность,упражнения
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?пр ?программа ?совпадение ?пациентПрограммы ?специалист ?длительностьНедель
       (COUNT(DISTINCT ?у) AS ?всегоУпражнений)
       (SUM(?плохое) AS ?неПодходит)
       (AVG(?эф) AS ?средняяЭффективность)
       (GROUP_CONCAT(?названиеУпр; SEPARATOR=", ") AS ?упражнения)
WHERE {
  VALUES ?пациент { reh:Пациент_Новиков }   # PARAM пациент
  ?пациент reh:имеетДанные/reh:содержитДиагноз ?диагноз ;
           reh:уровеньПодготовки ?подготовка ;
           reh:возраст ?возраст ;
           reh:весКг ?вес ;
           reh:ростСм ?рост .
  ?диагноз reh:требуетЦели ?нужнаяЦель .
  ?нужнаяЦель a ?видНужнойЦели .
  ?видНужнойЦели rdfs:subClassOf reh:УлучшениеКачестваЖизни .

  ?пр a reh:РеабилитационнаяПрограмма ;
      reh:название ?программа ;
      reh:имеетЦель ?цель ;
      reh:длительностьНедель ?длительностьНедель ;
      reh:предназначенаДля ?другой ;
      reh:основанаНаОпытеСпециалиста/reh:ФИО ?специалист .
  FILTER (?другой != ?пациент)
  ?другой reh:ФИО ?пациентПрограммы ;
          reh:имеетДанные/reh:содержитДиагноз ?егоДиагноз .
  ?цель a ?видЦели .
  ?видЦели rdfs:subClassOf reh:УлучшениеКачестваЖизни .
  BIND (IF(?егоДиагноз = ?диагноз, 3,
        IF(?цель = ?нужнаяЦель, 2,
        IF(?видЦели = ?видНужнойЦели, 1, 0))) AS ?совпадение)
  FILTER (?совпадение > 0)

  ?пр reh:содержитУпражнение ?у .
  ?у reh:название ?названиеУпр ;
     reh:уровеньСложности ?сложность ;
     reh:основаноНаИспытании/reh:показываетЭффективность/reh:процентЭффективности ?эф .
  BIND (?вес / ((?рост / 100.0) * (?рост / 100.0)) AS ?имт)
  BIND (IF(?имт >= 30 || ?возраст >= 60, 2, ?подготовка + 1) AS ?максСложность)
  BIND (IF(EXISTS { ?у reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз }
           || ?сложность > ?максСложность, 1, 0) AS ?плохое)
}
GROUP BY ?пр ?программа ?совпадение ?пациентПрограммы ?специалист ?длительностьНедель
ORDER BY DESC(?совпадение) ?неПодходит DESC(?средняяЭффективность)

# Файл: queries/recommend/04_specialist_for_patient.rq
# id: specialist_for_patient
# title: Врач или тренер для пациента
# button: Порекомендовать специалиста
# description: Специалисты, которые специализируются на диагнозе пациента. Если диагноз требует врача (травма, перелом, инсульт) — только реабилитологи. Оценка = рейтинг × 20 + стаж − 5 × число уже ведомых программ (меньше загрузка — выше оценка).
# param: пациент | individual:Пациент | reh:Пациент_Новиков
# card: title=специалист; score=оценка; fields=тип,рейтинг,стажЛет,нагрузка,диагноз
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?с ?специалист ?тип ?оценка ?рейтинг ?стажЛет ?нагрузка ?диагноз
WHERE {
  VALUES ?пациент { reh:Пациент_Новиков }   # PARAM пациент
  ?пациент reh:имеетДанные/reh:содержитДиагноз ?д .
  ?д reh:название ?диагноз ;
     reh:требуетВрача ?нуженВрач .
  ?с reh:специализируетсяНа ?д ;
     reh:ФИО ?специалист ;
     reh:рейтинг ?рейтинг ;
     reh:стажЛет ?стажЛет .
  BIND (EXISTS { ?с a reh:Реабилитолог } AS ?врач)
  FILTER (?нуженВрач = false || ?врач)
  BIND (IF(?врач, "Врач-реабилитолог", "Тренер ЛФК") AS ?тип)
  {
    SELECT ?с (COUNT(DISTINCT ?пр) AS ?нагрузка)
    WHERE { ?с reh:специализируетсяНа ?любой .
            OPTIONAL { ?пр reh:основанаНаОпытеСпециалиста ?с } }
    GROUP BY ?с
  }
  BIND (?рейтинг * 20 + ?стажЛет - 5 * ?нагрузка AS ?оценка)
}
ORDER BY DESC(?оценка)

# Файл: queries/recommend/05_equipment_for_patient.rq
# id: equipment_for_patient
# title: Спортивный инвентарь для пациента
# button: Порекомендовать инвентарь
# description: Инвентарь, который нужен для упражнений, рекомендованных пациенту (те же фильтры, что в «Упражнения для пациента»), в пределах бюджета. Сначала — то, что покрывает больше упражнений, затем — дешевле.
# param: пациент | individual:Пациент | reh:Пациент_Лебедева
# param: бюджет | number | 5000
# card: title=инвентарь; score=упражнений; fields=цена,производитель,страна,дляУпражнений,правила
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?инв ?инвентарь ?цена ?производитель ?страна ?правила
       (COUNT(DISTINCT ?у) AS ?упражнений)
       (GROUP_CONCAT(DISTINCT ?упражнение; SEPARATOR=", ") AS ?дляУпражнений)
WHERE {
  VALUES ?пациент { reh:Пациент_Лебедева }   # PARAM пациент
  VALUES ?бюджет { 5000 }                    # PARAM бюджет
  ?пациент reh:имеетДанные/reh:содержитДиагноз ?диагноз ;
           reh:уровеньПодготовки ?подготовка ;
           reh:возраст ?возраст ;
           reh:весКг ?вес ;
           reh:ростСм ?рост .
  {
    ?показание reh:показаноПри ?диагноз .
    ?у reh:имеетПоказание ?показание .
  } UNION {
    ?диагноз reh:требуетЦели ?цель .
    ?у reh:влияетНа ?цель .
  }
  ?у reh:название ?упражнение ;
     reh:уровеньСложности ?сложность ;
     reh:основаноНаИспытании/reh:имеетСтатус/reh:название ?статус ;
     reh:требуетИнвентарь ?инв .
  ?инв reh:название ?инвентарь ;
       reh:цена ?цена ;
       reh:имеетПроизводителя ?п ;
       reh:имеетПравилаПользования/reh:описание ?правила .
  ?п reh:название ?производитель ;
     reh:страна ?страна .
  FILTER NOT EXISTS { ?у reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз }
  BIND (?вес / ((?рост / 100.0) * (?рост / 100.0)) AS ?имт)
  BIND (IF(?имт >= 30 || ?возраст >= 60, 2, ?подготовка + 1) AS ?максСложность)
  FILTER (?сложность <= ?максСложность)
  FILTER (?статус != "Приостановлено")
  FILTER (?цена <= ?бюджет)
}
GROUP BY ?инв ?инвентарь ?цена ?производитель ?страна ?правила
ORDER BY DESC(?упражнений) ?цена

# Файл: queries/recommend/06_replace_exercise.rq
# id: replace_exercise
# title: Замена неподходящих упражнений в программе
# button: Найти замену упражнениям
# description: Проверяет текущую программу пациента. Для каждого упражнения, которое противопоказано при его диагнозе или слишком сложное, предлагает замену с тем же лечебным эффектом или той же целью: без противопоказаний, по силам пациенту и ещё не входящую в программу.
# param: пациент | individual:Пациент | reh:Пациент_Кузнецова
# card: title=замена; score=рейтингЗамены; warn=проблема; fields=проблемное,общее,сложностьЗамены
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?проблемное ?проблема ?зам ?замена ?рейтингЗамены ?сложностьЗамены
       (GROUP_CONCAT(DISTINCT ?связь; SEPARATOR="; ") AS ?общее)
WHERE {
  VALUES ?пациент { reh:Пациент_Кузнецова }   # PARAM пациент
  ?пр reh:предназначенаДля ?пациент ;
      reh:содержитУпражнение ?у .
  ?пациент reh:имеетДанные/reh:содержитДиагноз ?диагноз ;
           reh:уровеньПодготовки ?подготовка ;
           reh:возраст ?возраст ;
           reh:весКг ?вес ;
           reh:ростСм ?рост .
  ?у reh:название ?проблемное ;
     reh:уровеньСложности ?сложность .
  BIND (?вес / ((?рост / 100.0) * (?рост / 100.0)) AS ?имт)
  BIND (IF(?имт >= 30 || ?возраст >= 60, 2, ?подготовка + 1) AS ?максСложность)
  BIND (IF(EXISTS { ?у reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз },
           "Противопоказано при диагнозе пациента",
        IF(?сложность > ?максСложность, "Сложнее, чем позволяет состояние пациента", "")) AS ?проблема)
  FILTER (?проблема != "")

  {   # замена с тем же лечебным эффектом
    ?у reh:имеетЭффект ?эффект .
    ?зам reh:имеетЭффект ?эффект .
    ?эффект reh:описание ?связь .
  } UNION {   # замена, работающая на ту же цель
    ?у reh:влияетНа ?цель .
    ?зам reh:влияетНа ?цель .
    ?цель reh:название ?связь .
  }
  FILTER (?зам != ?у)
  ?зам reh:название ?замена ;
       reh:уровеньСложности ?сложностьЗамены ;
       reh:основаноНаИспытании ?и .
  ?и reh:показываетЭффективность/reh:процентЭффективности ?эф ;
     reh:определяетУровеньБезопасности/reh:баллБезопасности ?без ;
     reh:имеетСтатус/reh:название ?статус .
  FILTER NOT EXISTS { ?зам reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз }
  FILTER NOT EXISTS { ?пр reh:содержитУпражнение ?зам }
  FILTER (?сложностьЗамены <= ?максСложность)
  FILTER (?статус != "Приостановлено")
  BIND (0.6 * ?эф + 8 * ?без AS ?рейтингЗамены)
}
GROUP BY ?проблемное ?проблема ?зам ?замена ?рейтингЗамены ?сложностьЗамены
ORDER BY ?проблемное DESC(?рейтингЗамены)

# Файл: queries/recommend/07_audit_programs.rq
# id: audit_programs
# title: Проверка всех программ
# button: Проверить все программы
# description: Проходит по всем назначенным программам и находит проблемы: упражнение противопоказано при диагнозе пациента, сложность выше допустимой, испытание упражнения приостановлено. Для каждой проблемы — рекомендация, что сделать.
# card: title=программа; warn=проблема; fields=пациент,упражнение,рекомендация
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?пр ?программа ?пациент ?упражнение ?проблема ?рекомендация
WHERE {
  ?пр reh:название ?программа ;
      reh:предназначенаДля ?пац ;
      reh:содержитУпражнение ?у .
  ?пац reh:ФИО ?пациент ;
       reh:имеетДанные/reh:содержитДиагноз ?диагноз ;
       reh:уровеньПодготовки ?подготовка ;
       reh:возраст ?возраст ;
       reh:весКг ?вес ;
       reh:ростСм ?рост .
  ?у reh:название ?упражнение ;
     reh:уровеньСложности ?сложность ;
     reh:основаноНаИспытании/reh:имеетСтатус/reh:название ?статус .
  BIND (?вес / ((?рост / 100.0) * (?рост / 100.0)) AS ?имт)
  BIND (IF(?имт >= 30 || ?возраст >= 60, 2, ?подготовка + 1) AS ?максСложность)
  VALUES ?вид { 1 2 3 }
  BIND (IF(?вид = 1 && EXISTS { ?у reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз },
           "Противопоказано при диагнозе пациента",
        IF(?вид = 2 && ?сложность > ?максСложность, "Сложность выше допустимой",
        IF(?вид = 3 && ?статус = "Приостановлено", "Испытание упражнения приостановлено", ""))) AS ?проблема)
  FILTER (?проблема != "")
  BIND (IF(?вид = 1, "Убрать упражнение и подобрать замену",
        IF(?вид = 2, "Заменить на более лёгкое или снизить нагрузку",
                     "Назначать только под контролем специалиста")) AS ?рекомендация)
}
ORDER BY ?программа ?упражнение

# Файл: queries/recommend/08_similar_patients.rq
# id: similar_patients
# title: Похожие пациенты и их программы
# button: Что назначали похожим пациентам
# description: «Пациентам, похожим на вас, назначали…». Похожесть 2 — тот же диагноз, 1 — цель реабилитации того же вида. Показывает программу, специалиста и длительность у похожих пациентов.
# param: пациент | individual:Пациент | reh:Пациент_Фёдорова
# card: title=похожийПациент; score=похожесть; fields=егоДиагноз,программа,специалист,длительностьНедель,занятийВНеделю
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?другой ?похожийПациент ?похожесть ?егоДиагноз ?программа ?специалист ?длительностьНедель ?занятийВНеделю
WHERE {
  VALUES ?пациент { reh:Пациент_Фёдорова }   # PARAM пациент
  ?пациент reh:имеетДанные/reh:содержитДиагноз ?диагноз .
  ?диагноз reh:требуетЦели/a ?видЦели .
  ?видЦели rdfs:subClassOf reh:УлучшениеКачестваЖизни .

  ?другой a reh:Пациент ;
          reh:ФИО ?похожийПациент ;
          reh:имеетДанные/reh:содержитДиагноз ?д2 .
  FILTER (?другой != ?пациент)
  ?д2 reh:название ?егоДиагноз ;
      reh:требуетЦели/a ?видЦели2 .
  ?видЦели2 rdfs:subClassOf reh:УлучшениеКачестваЖизни .
  BIND (IF(?д2 = ?диагноз, 2, IF(?видЦели2 = ?видЦели, 1, 0)) AS ?похожесть)
  FILTER (?похожесть > 0)

  ?пр reh:предназначенаДля ?другой ;
      reh:название ?программа ;
      reh:длительностьНедель ?длительностьНедель ;
      reh:занятийВНеделю ?занятийВНеделю ;
      reh:основанаНаОпытеСпециалиста/reh:ФИО ?специалист .
}
ORDER BY DESC(?похожесть) ?похожийПациент

# Файл: queries/recommend/09_schedule_for_patient.rq
# id: schedule_for_patient
# title: Длительность и частота занятий
# button: Рекомендовать длительность
# description: Рекомендуемая длительность программы и число занятий в неделю — средние значения по программам с целью того же вида, что требует диагноз пациента (подзапрос с AVG).
# param: пациент | individual:Пациент | reh:Пациент_Лебедева
# card: title=цель; fields=видЦели,недель,занятийВНеделю,основаноНаПрограммах
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?цель ?видЦели ?недель ?занятийВНеделю ?основаноНаПрограммах
WHERE {
  VALUES ?пациент { reh:Пациент_Лебедева }   # PARAM пациент
  ?пациент reh:имеетДанные/reh:содержитДиагноз/reh:требуетЦели ?ц .
  ?ц reh:название ?цель ;
     a ?вид .
  ?вид rdfs:subClassOf reh:УлучшениеКачестваЖизни ;
       rdfs:label ?видЦели .
  {
    SELECT ?вид (AVG(?н) AS ?недель) (AVG(?з) AS ?занятийВНеделю) (COUNT(?пр) AS ?основаноНаПрограммах)
    WHERE {
      ?пр reh:имеетЦель ?цельПр ;
          reh:длительностьНедель ?н ;
          reh:занятийВНеделю ?з .
      ?цельПр a ?вид .
      ?вид rdfs:subClassOf reh:УлучшениеКачестваЖизни .
    }
    GROUP BY ?вид
  }
}

# Файл: queries/recommend/10_top_exercises_for_goal.rq
# id: top_exercises_for_goal
# title: Топ упражнений для цели
# button: Лучшие упражнения для цели
# description: Пять лучших по рейтингу упражнений, которые работают на выбранную цель реабилитации. Отмечены оптимальные (эффективность ≥ 80 % и безопасность ≥ 4) и упражнения с приостановленным испытанием.
# param: цель | individual:ЦельРеабилитации | reh:Цель_УменьшениеБолиВСпине
# card: title=упражнение; score=рейтинг; tag=отметка; fields=эффективность,безопасность,сложность,инвентарь
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?у ?упражнение ?рейтинг ?отметка ?эффективность ?безопасность ?сложность ?инвентарь
WHERE {
  VALUES ?цель { reh:Цель_УменьшениеБолиВСпине }   # PARAM цель
  ?у reh:влияетНа ?цель ;
     reh:название ?упражнение ;
     reh:уровеньСложности ?сложность ;
     reh:основаноНаИспытании ?и .
  ?и reh:показываетЭффективность/reh:процентЭффективности ?эффективность ;
     reh:определяетУровеньБезопасности/reh:баллБезопасности ?безопасность ;
     reh:имеетСтатус/reh:название ?статус .
  OPTIONAL { ?у reh:требуетИнвентарь/reh:название ?инвентарь }
  BIND (0.6 * ?эффективность + 8 * ?безопасность AS ?рейтинг)
  BIND (IF(?статус = "Приостановлено", "Испытание приостановлено",
        IF(?эффективность >= 80 && ?безопасность >= 4, "Оптимальное", "")) AS ?отметка)
}
ORDER BY DESC(?рейтинг)
LIMIT 5

# Файл: queries/recommend/11_waiting_list.rq
# id: waiting_list
# title: Очередь на подбор программы
# button: Кому нужна программа
# description: Пациенты, которым ещё не подобрана программа (FILTER NOT EXISTS), с целью, которую требует их диагноз, и списком подходящих специалистов (с учётом «требует врача»).
# card: title=пациент; fields=диагноз,нужнаЦель,подходящиеСпециалисты
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?пац ?пациент ?диагноз ?нужнаЦель
       (GROUP_CONCAT(DISTINCT ?специалист; SEPARATOR=", ") AS ?подходящиеСпециалисты)
WHERE {
  ?пац a reh:Пациент ;
       reh:ФИО ?пациент ;
       reh:имеетДанные/reh:содержитДиагноз ?д .
  FILTER NOT EXISTS { ?пр reh:предназначенаДля ?пац }
  ?д reh:название ?диагноз ;
     reh:требуетВрача ?нуженВрач ;
     reh:требуетЦели/reh:название ?нужнаЦель .
  OPTIONAL {
    ?с reh:специализируетсяНа ?д ;
       reh:ФИО ?специалист .
    FILTER (?нуженВрач = false || EXISTS { ?с a reh:Реабилитолог })
  }
}
GROUP BY ?пац ?пациент ?диагноз ?нужнаЦель
ORDER BY ?пациент

# Файл: queries/recommend/12_can_patient_do.rq
# id: can_patient_do
# title: Можно ли пациенту это упражнение?
# button: Проверить упражнение
# description: ASK-запрос: true, если упражнение подходит пациенту (по показанию или цели), не противопоказано при его диагнозе, по силам ему и испытание не приостановлено.
# param: пациент | individual:Пациент | reh:Пациент_Кузнецова
# param: упражнение | individual:Упражнение | reh:Упр_ЯгодичныйМостик
# card: boolean
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

ASK {
  VALUES ?пациент { reh:Пациент_Кузнецова }      # PARAM пациент
  VALUES ?у { reh:Упр_ЯгодичныйМостик }          # PARAM упражнение
  ?пациент reh:имеетДанные/reh:содержитДиагноз ?диагноз ;
           reh:уровеньПодготовки ?подготовка ;
           reh:возраст ?возраст ;
           reh:весКг ?вес ;
           reh:ростСм ?рост .
  { ?у reh:имеетПоказание/reh:показаноПри ?диагноз }
  UNION
  { ?диагноз reh:требуетЦели ?цель . ?у reh:влияетНа ?цель }
  ?у reh:уровеньСложности ?сложность ;
     reh:основаноНаИспытании/reh:имеетСтатус/reh:название ?статус .
  FILTER NOT EXISTS { ?у reh:имеетПротивопоказание/reh:противопоказаноПри ?диагноз }
  BIND (?вес / ((?рост / 100.0) * (?рост / 100.0)) AS ?имт)
  BIND (IF(?имт >= 30 || ?возраст >= 60, 2, ?подготовка + 1) AS ?максСложность)
  FILTER (?сложность <= ?максСложность)
  FILTER (?статус != "Приостановлено")
}

# Файл: queries/recommend/13_contraindicated_for_patient.rq
# id: contraindicated_for_patient
# title: Противопоказанные упражнения
# button: Что мне нельзя
# description: Упражнения, которые пациенту выполнять нельзя или только с осторожностью, и почему: противопоказание при его диагнозе, сложность выше уровня подготовки + 1, сложность выше 2 при ожирении (ИМТ ≥ 30) или в возрасте 60+, приостановленное клиническое испытание.
# param: пациент | individual:Пациент | reh:Пациент_Кузнецова
# card: title=упражнение; warn=причины; fields=подробности,сложность,инвентарь
PREFIX reh: <http://www.semanticweb.org/rehab/ontologies/2026/rehab-program#>

SELECT ?у ?упражнение ?сложность ?инвентарь
       (GROUP_CONCAT(DISTINCT ?причина; SEPARATOR="; ") AS ?причины)
       (GROUP_CONCAT(DISTINCT ?подробность; SEPARATOR="; ") AS ?подробности)
WHERE {
  VALUES ?пациент { reh:Пациент_Кузнецова }   # PARAM пациент
  {   # 1. противопоказание при диагнозе пациента
    ?пациент reh:имеетДанные/reh:содержитДиагноз ?диагноз .
    ?у reh:имеетПротивопоказание ?противопоказание .
    ?противопоказание reh:противопоказаноПри ?диагноз ;
                      reh:описание ?подробность .
    BIND ("Противопоказано при диагнозе" AS ?причина)
  } UNION {   # 2. сложность выше уровня подготовки + 1
    ?пациент reh:уровеньПодготовки ?подготовка .
    ?у a reh:Упражнение ;
       reh:уровеньСложности ?с2 .
    FILTER (?с2 > ?подготовка + 1)
    BIND ("Слишком сложно для уровня подготовки" AS ?причина)
    BIND ("Сложность выше уровня подготовки + 1" AS ?подробность)
  } UNION {   # 3. ожирение (ИМТ ≥ 30) — только сложность ≤ 2
    ?пациент reh:весКг ?вес ;
             reh:ростСм ?рост .
    ?у a reh:Упражнение ;
       reh:уровеньСложности ?с3 .
    FILTER (?вес / ((?рост / 100.0) * (?рост / 100.0)) >= 30 && ?с3 > 2)
    BIND ("Ожирение: допустима сложность не выше 2" AS ?причина)
    BIND ("ИМТ ≥ 30" AS ?подробность)
  } UNION {   # 4. возраст 60+ — только сложность ≤ 2
    ?пациент reh:возраст ?возраст .
    ?у a reh:Упражнение ;
       reh:уровеньСложности ?с4 .
    FILTER (?возраст >= 60 && ?с4 > 2)
    BIND ("Возраст 60+: допустима сложность не выше 2" AS ?причина)
    BIND ("Возраст 60 лет и старше" AS ?подробность)
  } UNION {   # 5. клиническое испытание приостановлено
    ?у reh:основаноНаИспытании/reh:имеетСтатус reh:Статус_Приостановлено .
    BIND ("Испытание приостановлено — только с осторожностью" AS ?причина)
    BIND ("Клиническое испытание упражнения приостановлено" AS ?подробность)
  }
  ?у reh:название ?упражнение ;
     reh:уровеньСложности ?сложность .
  OPTIONAL { ?у reh:требуетИнвентарь/reh:название ?инвентарь }
}
GROUP BY ?у ?упражнение ?сложность ?инвентарь
ORDER BY DESC(?сложность) ?упражнение
